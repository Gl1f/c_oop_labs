cmake_minimum_required(VERSION 3.16)
project(Matrix LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This program uses the Windows console API")
endif()
add_executable(matrix src/main.cpp src/Application.cpp src/Console.cpp src/Line.cpp src/Symbol.cpp)
target_compile_features(matrix PRIVATE cxx_std_11)
target_link_libraries(matrix PRIVATE user32)
if(MSVC)
    target_compile_options(matrix PRIVATE /W4 /utf-8)
else()
    target_compile_options(matrix PRIVATE -Wall -Wextra -pedantic)
endif()
